Using the :has() relational selector Jump to heading#

How to use the

() relational selector to make a star-rating component

CodePen

<fieldset class="star-rating">
    <legend>Rating</legend>
    <input type="radio" name="rating" value="1" id="rating-1" />
    <label for="rating-1">1</label>
    <input type="radio" name="rating" value="2" id="rating-2" />
    <label for="rating-2">2</label>
    <input type="radio" name="rating" value="3" id="rating-3" />
    <label for="rating-3">3</label>
    <input type="radio" name="rating" value="4" id="rating-4" />
    <label for="rating-4">4</label>
    <input type="radio" name="rating" value="5" id="rating-5" />
    <label for="rating-5">5</label>
</fieldset>
.star-rating {
    all: unset;
}

.star-rating legend,
.star-rating label {
    display: none;
}

.star-rating [type="radio"] {
    all: unset;
    cursor: pointer;
    font-size: 4rem;
    line-height: 1;
}

.star-rating [type="radio"]:checked,
.star-rating [type="radio"]:has(~ :checked) {
    --icon: "★";
}

.star-rating [type="radio"]::before {
    content: var(--icon, "☆");
}

.star-rating [type="radio"]:focus-visible {
    outline: 5px auto -webkit-focus-ring-color;
}
